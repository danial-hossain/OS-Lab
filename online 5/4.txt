Semaphore Empty = 5;
Semaphore Full = 0;
Semaphore S = 1;

int In = 0;
int Out = 0;
int Buffer[5];


Producer()
{
    for(i = 1; i <= 20; i++)
    {
        Down(Empty);
        Down(S);

        Buffer[In] = i;
        In = (In + 1) mod 5;

        Up(S);
        Up(Full);
    }
}


Consumer1()       // Even
{
    int count = 0;

    while(count < 10)
    {
        Down(Full);
        Down(S);

        Item = Buffer[Out];

        if(Item % 2 == 0)
        {
            Consume Item;
            Out = (Out + 1) mod 5;
            count++;

            Up(S);
            Up(Empty);
        }
        else
        {
            Up(S);
            Up(Full);
        }
    }
}


Consumer2()       // Odd
{
    int count = 0;

    while(count < 10)
    {
        Down(Full);
        Down(S);

        Item = Buffer[Out];

        if(Item % 2 != 0)
        {
            Consume Item;
            Out = (Out + 1) mod 5;
            count++;

            Up(S);
            Up(Empty);
        }
        else
        {
            Up(S);
            Up(Full);
        }
    }
}


main()
{
    Create 1 Producer thread;
    Create 2 Consumer threads;

    Wait for all threads;

    End;
}
